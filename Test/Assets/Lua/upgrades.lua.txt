--- 随机选 count 个不重复的升级
--- @param count number 选择的数量
--- @return table 选中升级的列表
function M.getRandomUpgrades(count)
    local pool = {}
    for _, v in ipairs(M.upgrades) do
        pool[#pool + 1] = v
    end

    -- Fisher-Yates 洗牌
    for i = #pool, 2, -1 do
        local j = math.random(1, i)
        pool[i], pool[j] = pool[j], pool[i]
    end

    local result = {}
    for i = 1, math.min(count, #pool) do
        result[i] = pool[i]
    end
    return result
end

--- 获取升级总数
--- @return number
function M.getUpgradeCount()
    return #M.upgrades
end

return M
